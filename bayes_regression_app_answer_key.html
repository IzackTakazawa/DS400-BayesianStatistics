# Simulate directly from the priors (Exponential(l) == Gamma(1, l), so
# rexp() and bayesrules::plot_gamma(shape=1, rate=l) describe the same
# sigma prior -- this reuses the book's own plotting functions.
simulate_prior_draws <- function(m0, s0, m1, s1, l, n = 200) {
  tibble(
    beta0c = rnorm(n, m0, s0),
    beta1  = rnorm(n, m1, s1),
    sigma  = rexp(n, l)
  )
}

# beta0c is the CENTERED intercept (typical ridership at the average
# temperature) -- to draw a line against raw temp_feel we decenter it,
# exactly the beta0 vs beta0c distinction from the book's Figure 9.3.
prior_line_df <- function(draws, x_seq) {
  draws %>%
    mutate(draw_id = row_number()) %>%
    crossing(x = x_seq) %>%
    mutate(mu = beta0c + beta1 * (x - mean_temp))
}

stan_call_text <- function(m0, s0, m1, s1, l, prior_pd, chains, iter) {
  sprintf(
    paste0(
      "stan_glm(\n",
      "  rides ~ temp_feel, data = bikes,\n",
      "  family = gaussian,\n",
      "  prior_intercept = normal(%s, %s),\n",
      "  prior = normal(%s, %s),\n",
      "  prior_aux = exponential(%s),\n",
      "  chains = %d, iter = %d, seed = 84735%s)"
    ),
    m0, s0, m1, s1, l, chains, iter,
    if (prior_pd) ",\n  prior_PD = TRUE" else ""
  )
}

# One reference card per question: a tab badge and the prompt, no
# answer box -- answers go in the comment block at the top of this
# script instead, so they persist with the file rather than the app.
question_card <- function(n, tab, prompt) {
  div(class = "question-card",
      h5(sprintf("Question %d", n), tags$span(class = "question-tab-badge", tab)),
      p(prompt)
  )
}

server <- function(input, output, session) {
  
  ## ===== TAB 1: Prior tuning (instant, no MCMC) =====
  
  observeEvent(input$reset_prior, {
    updateSliderInput(session, "m0", value = 5000)
    updateSliderInput(session, "s0", value = 1000)
    updateSliderInput(session, "m1", value = 100)
    updateSliderInput(session, "s1", value = 40)
    updateSliderInput(session, "l", value = 0.0008)
  })
  
  prior_draws <- reactive({
    simulate_prior_draws(input$m0, input$s0, input$m1, input$s1, input$l, n = 200)
  })
  
  output$priorStatement <- renderUI({
    m0 <- input$m0; s0 <- input$s0; m1 <- input$m1; s1 <- input$s1; l <- input$l
    div(class = "interp-panel",
        sprintf(
          paste0(
            "Before seeing any data: on a typical (~%.0f\u00b0F) day, you expect ",
            "about %.0f riders (95%% plausible range roughly %.0f to %.0f). For ",
            "every one-degree increase in temperature, you expect ridership to ",
            "change by about %.0f rides (95%% plausible range roughly %.0f to ",
            "%.0f). Day-to-day scatter around that average is expected to be ",
            "about %.0f rides (E(sigma) = 1/l)."
          ),
          mean_temp, m0, m0 - 1.96*s0, m0 + 1.96*s0,
          m1, m1 - 1.96*s1, m1 + 1.96*s1,
          1/l
        )
    )
  })
  
  output$beta0Plot <- renderPlot({
    plot_normal(mean = input$m0, sd = input$s0) + labs(x = "beta0c", y = "pdf") + theme_minimal()
  })
  output$beta1Plot <- renderPlot({
    plot_normal(mean = input$m1, sd = input$s1) + labs(x = "beta1", y = "pdf") + theme_minimal()
  })
  output$sigmaPlot <- renderPlot({
    plot_gamma(shape = 1, rate = input$l) + labs(x = "sigma", y = "pdf") + theme_minimal()
  })
  
  output$priorLinesPlot <- renderPlot({
    x_seq <- seq(temp_range[1], temp_range[2], length.out = 50)
    prior_line_df(prior_draws(), x_seq) %>%
      ggplot(aes(x = x, y = mu, group = draw_id)) +
      geom_line(alpha = 0.15, color = "#4C72B0") +
      labs(x = "temp_feel", y = "rides",
           title = "200 prior plausible relationships between temp_feel and rides") +
      theme_minimal(base_size = 13)
  })
  
  output$priorCode <- renderText({
    stan_call_text(input$m0, input$s0, input$m1, input$s1, input$l,
                   prior_pd = TRUE, chains = 4, iter = 2000)
  })
  
  ## ===== TAB 2: Real posterior fit (MCMC, behind a button) =====
  
  output$currentPriorSummary <- renderText({
    sprintf("beta0c ~ Normal(%s, %s)\nbeta1  ~ Normal(%s, %s)\nsigma  ~ Exponential(%s)",
            input$m0, input$s0, input$m1, input$s1, input$l)
  })
  
  fitted_model <- reactiveVal(NULL)
  fitted_draws <- reactiveVal(NULL)
  
  observeEvent(input$fit_posterior, {
    withProgress(message = "Running MCMC (stan_glm)...", value = 0.3, {
      m <- stan_glm(
        rides ~ temp_feel, data = bikes_data, family = gaussian,
        prior_intercept = normal(input$m0, input$s0),
        prior = normal(input$m1, input$s1),
        prior_aux = exponential(input$l),
        chains = 4, iter = 2000, seed = 84735, refresh = 0
      )
      incProgress(0.6)
      fitted_model(m)
      fitted_draws(as.data.frame(m))
    })
  })
  
  output$modelFitted <- reactive({ !is.null(fitted_model()) })
